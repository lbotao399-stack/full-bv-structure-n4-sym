\begin{remark}[Physics notation: Clifford relation, Gamma-pairing and Dirac operator]
In the two-component notation of the physics notation key after Section 1.2, the Clifford relation along $V$ and along $W_{\mathbb R}^\perp$ reads
\[
\sigma^m\bar\sigma^n+\sigma^n\bar\sigma^m=-2\delta^{mn},\qquad u^{ik}w_{kj}+w^{ik}u_{kj}=-2(u,w)\,\delta^i_j\quad(u,w\in W_{\mathbb R}^\perp),
\]
and the $\Gamma$-pairing on $\Sigma$ and on $\Sigma^*$ has the components
\[
\begin{aligned}
&\Gamma(\xi_1,\xi_2)\big|_V^{\,m}=i\big(\bar\xi_{2\dot\alpha i}\bar\sigma^{m\dot\alpha\alpha}\xi_{1\alpha}{}^i+\bar\xi_{1\dot\alpha i}\bar\sigma^{m\dot\alpha\alpha}\xi_{2\alpha}{}^i\big),\\
&\Gamma(\xi_1,\xi_2)_{kl}=2\varepsilon_{jikl}\xi_{1\alpha}{}^j\varepsilon^{\alpha\beta}\xi_{2\beta}{}^i-2\big(\bar\xi_{1\dot\alpha k}\varepsilon^{\dot\alpha\dot\beta}\bar\xi_{2\dot\beta l}-\bar\xi_{1\dot\alpha l}\varepsilon^{\dot\alpha\dot\beta}\bar\xi_{2\dot\beta k}\big),\\
&\Gamma(\eta_1,\eta_2)\big|_V^{\,m}=-i\big(\eta_2{}^\alpha{}_i\sigma^m_{\alpha\dot\alpha}\bar\eta_1^{\dot\alpha i}+\eta_1{}^\alpha{}_i\sigma^m_{\alpha\dot\alpha}\bar\eta_2^{\dot\alpha i}\big),\\
&\Gamma(\eta_1,\eta_2)^{kl}=-2\varepsilon^{klpj}\varepsilon_{\alpha\beta}\eta_1{}^\beta{}_j\eta_2{}^\alpha{}_p+2\big(\varepsilon_{\dot\beta\dot\alpha}\bar\eta_1^{\dot\alpha l}\bar\eta_2^{\dot\beta k}-\varepsilon_{\dot\beta\dot\alpha}\bar\eta_1^{\dot\alpha k}\bar\eta_2^{\dot\beta l}\big),
\end{aligned}
\]
with $\Gamma_{kl}=\Gamma^I\Sigma^I_{kl}$, $\Gamma^{kl}=\tfrac12\varepsilon^{klij}\Gamma_{ij}$ and $(\Gamma,u)=\Gamma^mu_m+\tfrac14\Gamma_{ij}u^{ij}$. On two-forms along $V$,
\[
\rho(q(e_m\wedge e_n))\colon\quad \xi_\alpha{}^i\mapsto-\tfrac12(\sigma^m\bar\sigma^n-\sigma^n\bar\sigma^m)_\alpha{}^\beta\xi_\beta{}^i,\qquad \bar\xi_{\dot\alpha i}\mapsto\tfrac12\bar\xi_{\dot\beta i}(\bar\sigma^m\sigma^n-\bar\sigma^n\sigma^m)^{\dot\beta}{}_{\dot\alpha}.
\]
Further $d=dx^m\partial_m$, $\mathrm{dvol}=d^4x$, $(d^*dA)_n=-\partial_m\partial_mA_n+\partial_n\partial_mA_m$ on one-forms, and the Dirac operators are
\[
\begin{aligned}
&({\not d}\xi)^\alpha{}_i=i\,\partial_m\bar\xi_{\dot\alpha i}\bar\sigma^{m\dot\alpha\alpha},\qquad ({\not d}\xi)^{\dot\alpha i}=-i\,\bar\sigma^{m\dot\alpha\alpha}\partial_m\xi_\alpha{}^i,\\
&({\not d}|_{\Sigma^*}\eta)_\alpha{}^i=-i\,\sigma^m_{\alpha\dot\alpha}\partial_m\bar\eta^{\dot\alpha i},\qquad ({\not d}|_{\Sigma^*}\eta)_{\dot\alpha i}=i\,\partial_m\eta^\alpha{}_i\sigma^m_{\alpha\dot\alpha}.
\end{aligned}
\]
\end{remark}

\begin{remark}[Physics notation: four-dimensional chirality]
$\mathrm{Spin}(4,\mathbb C)=SL(2,\mathbb C)\times SL(2,\mathbb C)$ is the complexification of $SU(2)_L\times SU(2)_R$, and $\mathrm{Spin}(6,\mathbb C)\cong SL(4,\mathbb C)$ that of $SU(4)_R$, with $W=S_+^6$ the $\mathbf 4$. The eigenspaces of Proposition \ref{prop:c1-chirality} are
\[
+1:\ \ \xi_\alpha{}^i\in S_+\otimes W,\quad \eta^\alpha{}_i\in S_+\otimes W^*;\qquad -1:\ \ \bar\xi_{\dot\alpha i}\in S_-\otimes W^*,\quad \bar\eta^{\dot\alpha i}\in S_-\otimes W .
\]
On the Dirac column $(\xi_\alpha{}^i,\bar\eta^{\dot\alpha i})$ at fixed $i$,
\[
\rho(e_m)=\begin{pmatrix}0&-i\sigma^m\\-i\bar\sigma^m&0\end{pmatrix},\qquad \rho(e_1)\rho(e_2)\rho(e_3)\rho(e_4)=-\gamma_{5S}=\gamma_{5W}=\mathrm{diag}(1,1,-1,-1),
\]
so the eigenvalue $+1$ is Srednicki's left-handed chirality and Weinberg's $\gamma_5=+1$. In indices, $\rho(v)$ exchanges undotted and dotted indices and keeps the position of the $SU(4)$ index, $\xi_\alpha{}^i\mapsto(\rho(v)\xi)^{\dot\alpha i}$; $\rho(w)$ keeps the spinor type and maps the $\mathbf 4$ to the $\bar{\mathbf 4}$ through $w_{ji}$, $\xi_\alpha{}^i\mapsto(\rho(w)\xi)^\alpha{}_j$, and the $\bar{\mathbf 4}$ to the $\mathbf 4$ through $w^{ji}$. The opposite orientation exchanges dotted and undotted indices in all four eigenspaces.
\end{remark}

\begin{remark}[Physics notation: components of the fields and antifields]
Under $SU(2)_L\times SU(2)_R\times SU(4)_R$:
\begin{itemize}
\item $c$: the Faddeev--Popov ghost, a singlet.
\item $A$: $A_M=(A_m,A_I)$, the gauge field $A_m\in(\mathbf 2,\mathbf 2,\mathbf 1)$ and the six scalars $A_I\in(\mathbf 1,\mathbf 1,\mathbf 6)$, with $A_{ij}=A_I\Sigma^I_{ij}$ and $A^{ij}=\tfrac12\varepsilon^{ijkl}A_{kl}$.
\item $\lambda$: $\lambda_\alpha{}^i\in(\mathbf 2,\mathbf 1,\mathbf 4)$ and $\bar\lambda_{\dot\alpha i}\in(\mathbf 1,\mathbf 2,\bar{\mathbf 4})$.
\item Antifields: $A^*_M=(A^*_m,A^*_I)$ with $A^*_{ij}=A^*_I\Sigma^I_{ij}$, $A^{*ij}=\tfrac12\varepsilon^{ijkl}A^*_{kl}$; $\lambda^{*\alpha}{}_i\in(\mathbf 2,\mathbf 1,\bar{\mathbf 4})$ and $\bar\lambda^{*\dot\alpha i}\in(\mathbf 1,\mathbf 2,\mathbf 4)$; $c^*$, a singlet.
\end{itemize}
The identifications of duals give the fibre pairings, with $(-,-)$ on $\mathfrak g$ kept on the colour factors,
\[
\begin{aligned}
&(A,A^*)=(A_m,A^*_m)+(A_I,A^*_I)=(A_m,A^*_m)+\tfrac14(A^{ij},A^*_{ij}),\\
&(\lambda,\lambda^*)=(\lambda_\alpha{}^i,\lambda^{*\alpha}{}_i)-(\bar\lambda_{\dot\alpha i},\bar\lambda^{*\dot\alpha i}),\qquad (c^*,c).
\end{aligned}
\]
\end{remark}

\begin{remark}[Physics notation: the BV symplectic form]
With $\mathrm{dvol}=d^4x$,
\[
\begin{aligned}
&\langle A,A^*\rangle=\int d^4x\,\big[(A_m,A^*_m)+(A_I,A^*_I)\big],\qquad \langle c^*,c\rangle=\int d^4x\,(c^*,c),\\
&\langle\lambda^*,\lambda\rangle=\int d^4x\,\big[(\lambda_\alpha{}^i,\lambda^{*\alpha}{}_i)-(\bar\lambda_{\dot\alpha i},\bar\lambda^{*\dot\alpha i})\big],
\end{aligned}
\]
and $\langle A^*,A\rangle$, $\langle\lambda,\lambda^*\rangle$, $\langle c,c^*\rangle$ are their negatives. The field--antifield pairs are $(A_m,A^*_m)$, $(A_I,A^*_I)$, $(\lambda_\alpha{}^i,\lambda^{*\alpha}{}_i)$, $(\bar\lambda_{\dot\alpha i},\bar\lambda^{*\dot\alpha i})$, the last with the sign $-1$, and $(c,c^*)$; the even members $A_M$, $\lambda^{*\alpha}{}_i$, $\bar\lambda^{*\dot\alpha i}$, $c^*$ are the Grassmann-even coordinates.
\end{remark}

\begin{remark}[Physics notation: kernel of the identity, antibracket and BV Laplacian]
With $\delta(x-y)=\delta^4(x-y)$, the components of $K_0$ are $-\delta(x-y)\,\delta_{MN}(-,-)^{-1}$ on $A_M\otimes A^*_N$ and $A^*_N\otimes A_M$, $-\delta(x-y)\,(-,-)^{-1}$ on $c\otimes c^*$ and $c^*\otimes c$, and on $\lambda\otimes\lambda^*$ and $\lambda^*\otimes\lambda$ $-\delta(x-y)\,(-,-)^{-1}$ times the canonical element of $\Sigma\otimes\Sigma^*$, whose components are $\delta_\alpha^\beta\delta^i_j$ between $\lambda_\alpha{}^i$ and $\lambda^{*\beta}{}_j$ and $-\delta_{\dot\alpha}^{\dot\beta}\delta_i^j$ between $\bar\lambda_{\dot\alpha i}$ and $\bar\lambda^{*\dot\beta j}$. Further, $\{-,-\}$ is the BV antibracket and $\Delta_0=-\partial_{K_0}$ the BV Laplacian, which contracts a field with its antifield at coincident points.
\end{remark}

\begin{remark}[Physics notation: the action in components]
With $D_m=\partial_m+[A_m,-]$, the components of $F_A=dA+\tfrac12[A\wedge A]$ are
\[
F_{mn}=\partial_mA_n-\partial_nA_m+[A_m,A_n],\qquad (F_A)_{mI}=D_mA_I,\qquad (F_A)_{IJ}=[A_I,A_J],
\]
and $(F_A,F_A)=\sum_{M<N}(F_{MN},F_{MN})$, so that
\[
-\tfrac14(F_A,F_A)=-\tfrac18(F_{mn},F_{mn})-\tfrac14(D_mA_I,D_mA_I)-\tfrac18([A_I,A_J],[A_I,A_J]),
\]
summed over $m,n,I,J$. The density of \eqref{eq:c1-action} is
\[
\begin{aligned}
S=\int d^4x\Big\{&-\tfrac18(F_{mn},F_{mn})-\tfrac1{16}(D_mA^{ij},D_mA_{ij})-\tfrac1{128}\big([A^{ij},A^{kl}],[A_{ij},A_{kl}]\big)\\
&+\tfrac i2\big(\lambda_\alpha{}^i,\bar\sigma^{m\dot\alpha\alpha}D_m\bar\lambda_{\dot\alpha i}\big)+\tfrac i2\big(\bar\lambda_{\dot\alpha i},\bar\sigma^{m\dot\alpha\alpha}D_m\lambda_\alpha{}^i\big)\\
&+\tfrac12\big(\lambda_\alpha{}^j,\varepsilon^{\alpha\beta}[A_{ji},\lambda_\beta{}^i]\big)-\tfrac12\big(\bar\lambda_{\dot\alpha j},\varepsilon^{\dot\alpha\dot\beta}[A^{ji},\bar\lambda_{\dot\beta i}]\big)\\
&+(D_mc,A^*_m)+\tfrac14([A^{ij},c],A^*_{ij})+([c,\lambda_\alpha{}^i],\lambda^{*\alpha}{}_i)-([c,\bar\lambda_{\dot\alpha i}],\bar\lambda^{*\dot\alpha i})+\tfrac12([c,c],c^*)\Big\}.
\end{aligned}
\]
For $\mathfrak{su}(N)$ in the normalisation of the key, with $[T^a,T^b]=if^{abc}T^c$, $A^a_{ij}=A^a_I\Sigma^I_{ij}$, $A^{a\,ij}=\tfrac12\varepsilon^{ijkl}A^a_{kl}$ and Srednicki's contractions, the scalar potential and the Yukawa terms of $S/\hbar$ are
\[
-\tfrac{g^2}4f^{abe}f^{cde}A^a_IA^b_JA^c_IA^d_J-\tfrac g2f^{abc}A^b_{ji}\big(\lambda^{aj}\lambda^{ci}\big)-\tfrac g2f^{abc}A^{b\,ji}\big(\bar\lambda^a_j\bar\lambda^c_i\big).
\]
\end{remark}

\begin{remark}[Physics notation: free part, Fourier convention and real slice]
Up to total derivatives, \eqref{eq:c1-free} is
\[
\tfrac12\langle e,Qe\rangle=\int d^4x\Big\{-\tfrac18(F^0_{mn},F^0_{mn})-\tfrac14(\partial_mA_I,\partial_mA_I)+i\big(\bar\lambda_{\dot\alpha i},\bar\sigma^{m\dot\alpha\alpha}\partial_m\lambda_\alpha{}^i\big)+(\partial_mc,A^*_m)\Big\},
\]
with $F^0_{mn}=\partial_mA_n-\partial_nA_m$.
The coefficient $-\tfrac14$ of Proposition \ref{prop:c1-norm} gives $-\tfrac18(F_{mn},F_{mn})$, which for $\mathfrak{su}(N)$ is $-\tfrac14F^a_{mn}F^a_{mn}$ in $S/\hbar$; $S^{(1)}(Q_1)$ is the supersymmetry term of the equivariant action, in components at the end of Section 2.3. The Fourier convention is $e^{i(p,x)}=e^{ip_mx_m}$, so $\partial_m\mapsto ip_m$ and $|p|^2=p_mp_m$. On the real slice of Remark \ref{rem:c1-real} the components $A^a_m$, $A^a_I$ of $A_M=-igA^a_MT^a$ are real and the complex conjugate of $A^a_{ij}$ is $A^{a\,ij}$; the eigenvalue $-\tfrac12|p|^2$ belongs to the three transverse components of $A_m$ and the six $A_I$, the eigenvalue $0$ to $A_m\propto p_m$.
\end{remark}

\begin{remark}[Physics notation: the linearised BRST operator]
With the slot of the image in brackets and $\partial^2=\partial_m\partial_m$,
\[
c\mapsto-\partial_mc\ [A_m],\quad A_m\mapsto\tfrac12(\partial^2A_m-\partial_m\partial_nA_n)\ [A^*_m],\quad A_I\mapsto\tfrac12\partial^2A_I\ [A^*_I],\quad A^*_m\mapsto-\partial_mA^*_m\ [c^*],
\]
\[
\lambda_\alpha{}^i\mapsto-i\,\bar\sigma^{m\dot\alpha\alpha}\partial_m\lambda_\alpha{}^i\ [\bar\lambda^{*\dot\alpha i}],\qquad \bar\lambda_{\dot\alpha i}\mapsto i\,\partial_m\bar\lambda_{\dot\alpha i}\bar\sigma^{m\dot\alpha\alpha}\ [\lambda^{*\alpha}{}_i],
\]
in momentum space with $\partial_m\mapsto ip_m$: the linearised gauge transformation $-\partial_mc$, the Maxwell and Klein--Gordon operators of $A_m$ and $A_I$, the Weyl operators of $\lambda_\alpha{}^i$ and $\bar\lambda_{\dot\alpha i}$, and the divergence $-\partial_mA^*_m$ in the $c^*$ slot.
\end{remark}

\begin{remark}[Physics notation: the vertices in components]
Summed over $m,n,I,J$, the terms of \eqref{eq:c1-I} are
\[
-\tfrac12\big(dA,\tfrac12[A\wedge A]\big)=-\tfrac12(\partial_mA_n,[A_m,A_n])-\tfrac12(\partial_mA_I,[A_m,A_I]),
\]
\[
-\tfrac14\big(\tfrac12[A\wedge A],\tfrac12[A\wedge A]\big)=-\tfrac18([A_m,A_n],[A_m,A_n])-\tfrac14([A_m,A_I],[A_m,A_I])-\tfrac18([A_I,A_J],[A_I,A_J]),
\]
\[
([A,c],A^*)=([A_m,c],A^*_m)+([A_I,c],A^*_I),\qquad ([c,\lambda],\lambda^*)=([c,\lambda_\alpha{}^i],\lambda^{*\alpha}{}_i)-([c,\bar\lambda_{\dot\alpha i}],\bar\lambda^{*\dot\alpha i}).
\]
In the vertex functions $\iota_pA(p_j)=p_mA_m(p_j)$, $[A(p_1),A(p_2)]=[A_n(p_1),A_n(p_2)]+[A_I(p_1),A_I(p_2)]$, and the $\lambda\lambda A$ vertex is the gauge coupling plus the Yukawa coupling,
\[
\begin{aligned}
\big(\lambda(p_1),\rho([A(p_3),\lambda(p_2)])\big)=\ &i\big(\lambda_\alpha{}^i(p_1),\bar\sigma^{m\dot\alpha\alpha}[A_m(p_3),\bar\lambda_{\dot\alpha i}(p_2)]\big)+i\big(\bar\lambda_{\dot\alpha i}(p_1),\bar\sigma^{m\dot\alpha\alpha}[A_m(p_3),\lambda_\alpha{}^i(p_2)]\big)\\
&+\big(\lambda_\alpha{}^j(p_1),\varepsilon^{\alpha\beta}[A_{ji}(p_3),\lambda_\beta{}^i(p_2)]\big)-\big(\bar\lambda_{\dot\alpha j}(p_1),\varepsilon^{\dot\alpha\dot\beta}[A^{ji}(p_3),\bar\lambda_{\dot\beta i}(p_2)]\big).
\end{aligned}
\]
The $r_j$ are Grassmann-odd sources for the legs $\lambda$, $c$ and $A^*$.
\end{remark}

\begin{remark}[Physics notation: gauge transformations and the master equation]
The gauge transformations of the proof of Proposition \ref{prop:c1-cme} are, in components,
\[
A_m\mapsto D_mc,\qquad A_I\mapsto[A_I,c],\qquad \lambda_\alpha{}^i\mapsto[c,\lambda_\alpha{}^i],\qquad \bar\lambda_{\dot\alpha i}\mapsto[c,\bar\lambda_{\dot\alpha i}],\qquad c\mapsto\tfrac12[c,c],
\]
the coefficients of $A^*_m$, $A^*_I$, $\lambda^{*\alpha}{}_i$, $-\bar\lambda^{*\dot\alpha i}$ and $c^*$ in the action; $\{S,S\}=0$ is the classical BV master equation (Zinn-Justin equation) and $QI+\tfrac12\{I,I\}=0$ its interacting part. The functionals of the proof are
\[
\int d^4x\,\big(\lambda,\rho([dc,\lambda])\big)=i\int d^4x\Big\{\big(\lambda_\alpha{}^i,\bar\sigma^{m\dot\alpha\alpha}[\partial_mc,\bar\lambda_{\dot\alpha i}]\big)+\big(\bar\lambda_{\dot\alpha i},\bar\sigma^{m\dot\alpha\alpha}[\partial_mc,\lambda_\alpha{}^i]\big)\Big\},
\]
\[
\int d^4x\,\big(df,\Gamma(s_1,s_2)\big)=\int d^4x\,\partial_mf\,\Gamma^m(s_1,s_2),\qquad \int d^4x\,([dc,c],A^*)=\int d^4x\,([\partial_mc,c],A^*_m).
\]
\end{remark}
