\begin{definition}[Wick map and Euclidean $\sigma$]\label{def:wick}
For $v\in V$ with Euclidean components $v^m$, $m=1,2,3,4$, the Lorentzian vector is $v^0=-iv^4$, $v^k=v^k$ ($k=1,2,3$); in particular $x^4=it$ and $\partial_0=i\partial_4$. Define
\[
\sigma_E^m{}_{\alpha\dot\alpha}=(\sigma^1,\sigma^2,\sigma^3,\,i\,1),\qquad
\bar\sigma_E^{m\dot\alpha\alpha}=(-\sigma^1,-\sigma^2,-\sigma^3,\,i\,1).
\]
Then
\[
\sigma^\mu v_\mu=\sigma_E^m v_m,\qquad \bar\sigma^\mu v_\mu=\bar\sigma_E^m v_m,\qquad
\sigma_E^m\bar\sigma_E^n+\sigma_E^n\bar\sigma_E^m=-2\delta^{mn},\qquad \bar\sigma_E^m\sigma_E^n+\bar\sigma_E^n\sigma_E^m=-2\delta^{mn},
\]
\[
\bar\sigma_E^{m\dot\alpha\alpha}=\varepsilon^{\alpha\beta}\varepsilon^{\dot\alpha\dot\beta}\sigma^m_{E\,\beta\dot\beta},\qquad \bar\sigma_E^m=-(\sigma_E^m)^\dagger .
\]
Below $\sigma^m\equiv\sigma^m_E$, $\bar\sigma^m\equiv\bar\sigma^m_E$, Euclidean indices up or down alike. Orientation of $V$: $dx^1\wedge dx^2\wedge dx^3\wedge dx^4$.
\end{definition}

\begin{definition}[$SU(4)$ blocks]\label{def:thooft}
$W\ni w^i$ (upper index), $W^*\ni w_i$ (lower index), $i=1,\dots,4$. On $W_{\mathbb R}^\perp$ with orthonormal basis $e_I$, $I=1,\dots,6$:
\[
\Sigma^I_{ij}=(\eta^1,\eta^2,\eta^3,i\bar\eta^1,i\bar\eta^2,i\bar\eta^3)_{ij},\qquad \bar\Sigma^{I}=(\Sigma^I)^{\dagger}=(-\eta^1,-\eta^2,-\eta^3,i\bar\eta^1,i\bar\eta^2,i\bar\eta^3),
\]
$\eta^a_{bc}=\bar\eta^a_{bc}=\epsilon_{abc}$ ($b,c\le3$), $\eta^a_{b4}=-\eta^a_{4b}=\delta_{ab}$, $\bar\eta^a_{b4}=-\bar\eta^a_{4b}=-\delta_{ab}$ (engine B, \texttt{thooft}). With $\varepsilon_{1234}=\varepsilon^{1234}=1$:

\]

\sum_I\Sigma^I_{ij}\Sigma^I_{kl}=2\varepsilon_{ijkl},\qquad \sum_I\bar\Sigma^{I\,ij}\bar\Sigma^{I\,kl}=2\varepsilon^{ijkl},\qquad \sum_I\bar\Sigma^{I\,ij}\Sigma^I_{kl}=-2\big(\delta^i_k\delta^j_l-\delta^i_l\delta^j_k\big).

\]
\end{definition}
