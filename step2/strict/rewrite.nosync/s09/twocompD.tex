\begin{remark}[Two-component forms along $\Delta$]
\label{rem:c9-twocompD}
With the components of Definition \ref{def:c8-twocomp}, $(\sigma^n\bar\sigma^m)_\alpha{}^\beta=\sigma^n_{\alpha\dot\alpha}\bar\sigma^{m\dot\alpha\beta}$ and $(\bar\sigma^m\sigma^n)^{\dot\beta}{}_{\dot\alpha}=\bar\sigma^{m\dot\beta\alpha}\sigma^n_{\alpha\dot\alpha}$,
\[
\begin{aligned}
&\int(\lambda^*,\lambda)=\int d^4x\Big\{\big(\lambda_\alpha{}^i,\lambda^{*\alpha}{}_i\big)-\big(\bar\lambda_{\dot\alpha i},\bar\lambda^{*\dot\alpha i}\big)\Big\},\qquad -\int(2E(\lambda)+3\lambda,\lambda^*)=-\int d^4x\Big\{\big((2E+3)\lambda_\alpha{}^i,\lambda^{*\alpha}{}_i\big)-\big((2E+3)\bar\lambda_{\dot\alpha i},\bar\lambda^{*\dot\alpha i}\big)\Big\},\\
&\int(\rho(x){\not d}\lambda,\lambda^*)=\int d^4x\Big\{-x_n(\sigma^n\bar\sigma^m)_\alpha{}^\beta\big(\partial_m\lambda_\beta{}^i,\lambda^{*\alpha}{}_i\big)+x_n\big(\partial_m\bar\lambda_{\dot\beta i}(\bar\sigma^m\sigma^n)^{\dot\beta}{}_{\dot\alpha},\bar\lambda^{*\dot\alpha i}\big)\Big\},\qquad \int(\rho(x)\lambda^*,\lambda^*)=-2i\int d^4x\,x_m\big(\lambda^{*\alpha}{}_i,\sigma^m_{\alpha\dot\alpha}\bar\lambda^{*\dot\alpha i}\big),
\end{aligned}
\]
and $\int(\lambda,{\not d}\lambda)$ as in Remark \ref{rem:c9-twocomp3}. Thus \eqref{eq:c9-ObDelta} contains $\tfrac{94}3\,i$ units times $\int d^4x\,(\bar\lambda_{\dot\alpha i},\bar\sigma^{m\dot\alpha\alpha}\partial_m\lambda_\alpha{}^i)$, the translation-invariant $\hbar F_1^{(1)}(\Delta)$ contains $-\tfrac{47}3$ units times $\int d^4x\{(\lambda_\alpha{}^i,\lambda^{*\alpha}{}_i)-(\bar\lambda_{\dot\alpha i},\bar\lambda^{*\dot\alpha i})\}$, and Weinberg's forms follow from $\rho(v)=\gamma_W^\mu v_\mu$ on the Dirac column. The sector parts of $S^{(1)}(P_b)$ in Theorem \ref{thm:c9-theta2} are $\tfrac14\int d^4x\,(b_m\partial_mA_{ij},A^{*ij})$, $-\int d^4x\big\{(b_m\partial_m\lambda_\alpha{}^i,\lambda^{*\alpha}{}_i)-(b_m\partial_m\bar\lambda_{\dot\alpha i},\bar\lambda^{*\dot\alpha i})\big\}$ and $\int d^4x\big\{(b_m\partial_mA_n,A^*_n)-(b_m\partial_mc,c^*)\big\}$, and the solution \eqref{eq:c9-F1Delta} at $\theta=\theta_\lambda=\theta_V=-\tfrac13$, $\kappa_V=\kappa_\lambda=0$, is
\[
\begin{aligned}
\hbar F_1^{(1)}(\Delta)=\ \text{units}\int d^4x\Big\{&\big((\tfrac{22}3-\tfrac13(2x_n\partial_n+2))A_m,A^*_m\big)+\tfrac14\big((\tfrac{22}3-\tfrac13(2x_n\partial_n+2))A_{ij},A^{*ij}\big)-\big((\tfrac{8}3-\tfrac23x_n\partial_n)c,c^*\big)\\
&-\big((\tfrac{47}3-\tfrac13(2x_n\partial_n+3))\lambda_\alpha{}^i,\lambda^{*\alpha}{}_i\big)+\big((\tfrac{47}3-\tfrac13(2x_n\partial_n+3))\bar\lambda_{\dot\alpha i},\bar\lambda^{*\dot\alpha i}\big)\Big\}.
\end{aligned}
\]
For $\xi=(\xi_\alpha{}^i,0)$ the functionals of Theorem \ref{thm:c9-theta2} (ii) are $\psi_1=i\int d^4x\,x_m\big(\bar\lambda_{\dot\alpha i}\bar\sigma^{m\dot\alpha\alpha}\xi_\alpha{}^i,c^*\big)$ and $\psi_2=\int d^4x\,x_m\big(A^*_m,\xi_\alpha{}^i\lambda^{*\alpha}{}_i\big)$, and $T_1,\dots,T_4$ are those of Remark \ref{rem:c11-twocomp}.
\end{remark}
