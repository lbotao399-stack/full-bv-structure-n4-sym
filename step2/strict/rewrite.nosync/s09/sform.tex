\begin{lemma}[Components of $s$ and the Wess--Zumino identities]
\label{lem:c9-sform}
Write $\Lambda^{(0)}=Q+\{I,-\}$ on local functionals and $S(x)=S^{(1)}(x)$, in the normalisation of Lemma \ref{lem:c2-components}. For a local cochain $F$ of degree $0$ and a local cochain $G$ of degree $1$,
\begin{equation}
\begin{aligned}
(sF)^{(1)}(x)&=\Lambda^{(0)}F^{(1)}(x)+\{F^{(0)},S(x)\},\\
(sF)^{(2)}(x,y)&=\Lambda^{(0)}F^{(2)}(x,y)+F^{(1)}([x,y])+(-1)^{|x|}\big(\{F^{(1)}(x),S(y)\}+\{S(x),F^{(1)}(y)\}\big)+\{F^{(0)},S^{(2)}(x,y)\},\\
(sG)^{(1)}(x)&=\Lambda^{(0)}G^{(1)}(x)+\{G^{(0)},S(x)\},\\
(sG)^{(2)}(x,y)&=\Lambda^{(0)}G^{(2)}(x,y)-G^{(1)}([x,y])+(-1)^{|x|}\{G^{(1)}(x),S(y)\}-\{S(x),G^{(1)}(y)\}+\{G^{(0)},S^{(2)}(x,y)\}.
\end{aligned}
\label{eq:c9-sform}
\end{equation}
Hence the one-loop equation $sF_1=\mathrm{Ob}$ of Theorem \ref{thm:c8-obstruction} (ii) reads, through arity two,
\begin{equation}
\begin{aligned}
&\Lambda^{(0)}F_1^{(0)}=\mathrm{Ob}^{(0)},\qquad \Lambda^{(0)}F_1^{(1)}(x)+\{F_1^{(0)},S(x)\}=\mathrm{Ob}^{(1)}(x),\\
&\Lambda^{(0)}F_1^{(2)}(x,y)+F_1^{(1)}([x,y])+(-1)^{|x|}\big(\{F_1^{(1)}(x),S(y)\}+\{S(x),F_1^{(1)}(y)\}\big)+\{F_1^{(0)},S^{(2)}(x,y)\}=\mathrm{Ob}^{(2)}(x,y),
\end{aligned}
\label{eq:c9-arity2}
\end{equation}
and $s\,\mathrm{Ob}=0$ (Theorem \ref{thm:c8-obstruction} (i)) gives the Wess--Zumino identities
\[
\begin{aligned}
&\Lambda^{(0)}\mathrm{Ob}^{(1)}(x)=-\{\mathrm{Ob}^{(0)},S(x)\},\\
&\Lambda^{(0)}\mathrm{Ob}^{(2)}(x,y)=\mathrm{Ob}^{(1)}([x,y])-(-1)^{|x|}\{\mathrm{Ob}^{(1)}(x),S(y)\}+\{S(x),\mathrm{Ob}^{(1)}(y)\}-\{\mathrm{Ob}^{(0)},S^{(2)}(x,y)\}.
\end{aligned}
\]
If $F_1^{(0)}$ and $F_1^{(1)}$ solve the equations of arity zero and one, the arity-two residual
\[
\mathcal R^{(2)}(x,y):=\mathrm{Ob}^{(2)}(x,y)-F_1^{(1)}([x,y])-(-1)^{|x|}\big(\{F_1^{(1)}(x),S(y)\}+\{S(x),F_1^{(1)}(y)\}\big)-\{F_1^{(0)},S^{(2)}(x,y)\}
\]
is $\Lambda^{(0)}$-closed, and the arity-two equation along $(x,y)$ is solvable exactly when its class in the cohomology $H^{-1}(\Lambda^{(0)})$ of the local functionals with its quantum numbers vanishes.
\end{lemma}

\begin{proof}
The formulas for $sF$ are the derivatives of the component formulas of Lemma \ref{lem:c2-components} along $S_{\mathfrak{psl}(4|4)}+tF$, since $\tfrac{d}{dt}\tfrac12\{f,f\}=\{f,\dot f\}$ for even $f$. For the bracket of Definition \ref{def:c1-bracket}, $\{f,g\}=\Delta_0(fg)-\Delta_0(f)g-(-1)^{|f|}f\Delta_0(g)$, a derivation $\mathcal D$ of degree $1$ that commutes with $\Delta_0$ up to sign ($\mathcal D=Q$, $\{I,-\}$, $\Lambda^{(0)}$, or $\{F,-\}$ with $|F|=0$) satisfies
\[
\mathcal D\{a,b\}=-\{\mathcal Da,b\}-(-1)^{|a|}\{a,\mathcal Db\},
\]
as in the proof of Lemma \ref{lem:c10-deriv}. The components of $sG$ are $\Lambda^{(0)}G^{(2)}+\alpha G^{(1)}([x,y])+\beta\{G^{(1)}(x),S(y)\}+\gamma\{S(x),G^{(1)}(y)\}+\delta\{G^{(0)},S^{(2)}(x,y)\}$ and $\Lambda^{(0)}G^{(1)}(x)+\delta_1\{G^{(0)},S(x)\}$ with signs fixed by $s^2=0$ on the exact cochains $G=sF$. For $F=F^{(0)}$, $\Lambda^{(0)}\{F^{(0)},S(x)\}=-\{\Lambda^{(0)}F^{(0)},S(x)\}$ since $\Lambda^{(0)}S(x)=0$, so $\delta_1=1$. For $F=F^{(1)}$, with $|S(x)|\equiv1+|x|$, $\Lambda^{(0)}\big(F^{(1)}([x,y])+(-1)^{|x|}(\{F^{(1)}(x),S(y)\}+\{S(x),F^{(1)}(y)\})\big)=\Lambda^{(0)}F^{(1)}([x,y])-(-1)^{|x|}\{\Lambda^{(0)}F^{(1)}(x),S(y)\}+\{S(x),\Lambda^{(0)}F^{(1)}(y)\}$, so $\alpha=\gamma=-1$ and $\beta=(-1)^{|x|}$. For $F=F^{(0)}$ in arity two, $\Lambda^{(0)}S^{(2)}(x,y)=-S([x,y])-(-1)^{|x|}\{S(x),S(y)\}$ (Lemma \ref{lem:c2-components}) and the derivation rule for $\{F^{(0)},-\}$ leave $(\delta-1)\{\Lambda^{(0)}F^{(0)},S^{(2)}(x,y)\}$, so $\delta=1$. The Wess--Zumino identities are $(s\,\mathrm{Ob})^{(1)}=(s\,\mathrm{Ob})^{(2)}=0$. For $G=\mathrm{Ob}-sF_1$, whose components of arity zero and one vanish, $(sG)^{(2)}=\Lambda^{(0)}G^{(2)}=\Lambda^{(0)}\mathcal R^{(2)}$ vanishes, and the arity-two equation is $\Lambda^{(0)}F_1^{(2)}(x,y)=\mathcal R^{(2)}(x,y)$.
\end{proof}
